#include <cstdio>
#include <array>
#include <string>

std::string HardwareManager::getGPU()
{
    std::array<char, 256> buffer;
    std::string result;

    FILE* pipe = popen(
        "lspci | grep -Ei 'vga|3d|display'",
        "r"
    );

    if(!pipe)
        return "UNKNOWN";

    while(fgets(buffer.data(), buffer.size(), pipe))
    {
        result += buffer.data();
    }

    pclose(pipe);

    if(result.empty())
        return "UNKNOWN";

    if(result.back() == '\n')
        result.pop_back();

    return result;
}
